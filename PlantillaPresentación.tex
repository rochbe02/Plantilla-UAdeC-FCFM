	\section{Código}
	
	\begin{frame}[fragile]{Ejemplo de código en Python}
		\begin{lstlisting}[language=Python]
			def factorial(n):
			# Caso base
			if n <= 1:
			return 1
			return n * factorial(n - 1)
			
			print(factorial(5))
		\end{lstlisting}
	\end{frame}
